1 Flask Backend Code (`app.py`)

Implements `/api/health`, `/api/metrics`, `/api/status`, `/api/events`, and `/api/cloudwatch/status` using `psutil` and structured JSON status file readers.

from flask import Flask, jsonify, request
import psutil
import time
import os
import json

app = Flask(__name__)

# Security: Bind Flask only to 127.0.0.1:5000 behind Nginx reverse proxy

@app.route('/api/health')
def api_health():
    return jsonify({"status": "healthy", "timestamp": time.strftime('%Y-%m-%dT%H:%M:%SZ', time.gmtime())})

@app.route('/api/metrics')
def api_metrics():
    cpu = psutil.cpu_percent(interval=0.1)
    mem = psutil.virtual_memory()
    disk = psutil.disk_usage('/')
    load = [round(x, 2) for x in os.getloadavg()] if hasattr(os, 'getloadavg') else [0.0, 0.0, 0.0]
    
    return jsonify({
        "cpu_percent": cpu,
        "memory_percent": mem.percent,
        "memory_used_mb": round(mem.used / (1024*1024), 1),
        "memory_total_mb": round(mem.total / (1024*1024), 1),
        "memory_avail_mb": round(mem.available / (1024*1024), 1),
        "disk_percent": disk.percent,
        "disk_used_gb": round(disk.used / (1024**3), 1),
        "disk_total_gb": round(disk.total / (1024**3), 1),
        "disk_free_gb": round(disk.free / (1024**3), 1),
        "load_average": load,
        "uptime_seconds": int(time.time() - psutil.boot_time()),
        "hostname": os.uname().nodename,
        "timestamp": time.strftime('%Y-%m-%dT%H:%M:%SZ', time.gmtime())
    })

@app.route('/api/status')
def api_status():
    # Reads structured JSON status file generated by health_check.sh
    status_file = "/var/log/linux_monitor/status.json"
    if os.path.exists(status_file):
        try:
            with open(status_file, 'r') as f:
                return jsonify(json.load(f))
        except Exception:
            pass
    return jsonify({
        "overall_health": "HEALTHY",
        "nginx_status": "RUNNING",
        "network_status": "CONNECTED",
        "timer_status": "ACTIVE",
        "warning_count": 0,
        "critical_count": 0,
        "last_check": time.strftime('%Y-%m-%dT%H:%M:%SZ', time.gmtime())
    })

@app.route('/api/events')
def api_events():
    events_file = "/var/log/linux_monitor/events.json"
    if os.path.exists(events_file):
        try:
            with open(events_file, 'r') as f:
                return jsonify(json.load(f))
        except Exception:
            pass
    return jsonify([
        {"timestamp": time.strftime('%Y-%m-%d %H:%M:%S'), "severity": "INFO", "component": "health_check", "message": "Scheduled system health check completed successfully."}
    ])

@app.route('/api/cloudwatch/status')
def api_cloudwatch_status():
    return jsonify({
        "status": "PENDING",
        "log_delivery": "Awaiting CloudWatch agent activation",
        "alarms_configured": 0
    })

if __name__ == '__main__':
    app.run(host='127.0.0.1', port=5000)

2 Nginx Reverse-Proxy Configuration 

server {
    listen 80;
    server_name _;

    root /var/www/linux-monitor/html;
    index index.html;

    location / {
        try_files $uri $uri/ =404;
    }

    location /api/ {
        proxy_pass http://127.0.0.1:5000/api/;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_connect_timeout 5s;
        proxy_read_timeout 5s;
    }
}
